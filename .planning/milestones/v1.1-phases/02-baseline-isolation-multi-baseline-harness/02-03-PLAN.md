```python
import random
from itertools import product  # NOT used here — no repetition dimension this phase (Phase 3 scope)

def build_execution_order(baselines: list[str], seed: int) -> list[str]:
    order = list(baselines)
    random.Random(seed).shuffle(order)   # instance-scoped RNG, never the shared `random` module
    return order
```
